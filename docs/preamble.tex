% Shared preamble for math.tex and bench.tex. Each document sets its own geometry
% before \input{preamble}. Notation that more than one part uses is defined here,
% once, with \newcommand, so a clash is a build error instead of a silent override;
% the per-part macro files only \providecommand their own symbols.

  decorations.pathmorphing,patterns}

\pgfplotsset{compat=1.18}
\usepgfplotslibrary{groupplots}
\usepackage{caption}
\captionsetup{justification=raggedright, singlelinecheck=true}
